// Ganti alamat ini dengan domain backend Anda di Vercel / Render / GitHub Pages const API_URL = "http://localhost:3000/api/jurnal"; // 1. Simpan Jurnal ke Neon Database via API async function handleFormSubmit(e) { e.preventDefault(); if (selectedJamList.length === 0) { alert("Silakan pilih Jam Ke (1-10) terlebih dahulu."); return; } const payload = { nama_guru: document.getElementById('namaGuru').value, mapel: document.getElementById('mapel').value, kelas: document.getElementById('pilihanKelas').value, jam_ke: selectedJamList.join(', '), materi: document.getElementById('materi').value, absensi: getAbsensiData(), foto_url: "", // Bisa diintegrasikan dengan Cloudinary/ImgBB jika ada upload gambar lokasi_gps: isLocationValid ? "Dalam Radius" : "Bypass GPS", latitude: userLat, longitude: userLng }; try { const response = await fetch(API_URL, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(payload) }); if (response.ok) { alert("Jurnal Mengajar Berhasil Disimpan ke Database Neon!"); resetForm(); renderRekapTable(); switchTab('rekapTab'); } else { alert("Gagal menyimpan data ke database."); } } catch (error) { console.error("Error:", error); alert("Terjadi kesalahan koneksi ke database."); } } // 2. Load Data Rekap dari Neon Database async function renderRekapTable() { const tbody = document.getElementById('rekapTableBody'); tbody.innerHTML = `Memuat data dari Neon DB...`; try { const res = await fetch(API_URL); const data = await res.json(); if (data.length === 0) { tbody.innerHTML = `Belum ada data jurnal di database.`; return; } tbody.innerHTML = data.map(item => ` ${new Date(item.waktu).toLocaleString('id-ID')} ${item.kelas} Jam ${item.jam_ke} ${item.mapel} ${item.materi} ${item.absensi} ${item.lokasi_gps} `).join(''); } catch (err) { tbody.innerHTML = `Gagal terhubung ke Database Neon.`; } } // 3. Hapus Data dari Neon Database async function deleteJurnal(id) { if (confirm("Hapus jurnal ini dari database Neon?")) { try { await fetch(`${API_URL}/${id}`, { method: 'DELETE' }); renderRekapTable(); } catch (err) { alert("Gagal menghapus data."); } } }
